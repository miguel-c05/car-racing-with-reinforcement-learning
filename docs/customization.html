 91    def get_optimal_line(self, iterations=200):
 92        """Calculates an optimal racing line using Laplacian smoothing on track center points.
 93
 94        The algorithm iteratively smooths the track path while constraining points to stay
 95        within the track boundaries. This creates a "racing line" that cuts corners and
 96        optimizes the path through the track.
 97
 98        Args:
 99            iterations (int, optional): Number of smoothing iterations. Higher values produce
100                smoother lines but may be computationally expensive. Defaults to 200.
101
102        Returns:
103            np.ndarray: An array of shape (N, 2) containing the (x, y) coordinates of the
104                optimal racing line, where N is the number of track points.
105        """
106        raw_track = self.env.unwrapped.track
107        path = np.array([[p[2], p[3]] for p in raw_track])
108        
109        # Keep the line well within track boundaries
110        SAFE_MARGIN = 1.5  # More conservative margin
111        max_displacement = TRACK_WIDTH - SAFE_MARGIN
112
113        optimized_path = np.copy(path)
114        num_points = len(path)
115
116        for iteration in range(iterations):
117            for i in range(num_points):
118                prev_idx = (i - 1) % num_points
119                next_idx = (i + 1) % num_points
120                
121                # Smoothing: pull towards midpoint of neighbors
122                midpoint = (optimized_path[prev_idx] + optimized_path[next_idx]) / 2
123                # Use more aggressive smoothing in later iterations
124                blend_factor = 0.3 if iteration < iterations // 2 else 0.5
125                optimized_path[i] = optimized_path[i] * (1 - blend_factor) + midpoint * blend_factor
126
127                # Constraint: enforce maximum distance from track center
128                center = path[i]
129                current = optimized_path[i]
130                diff = current - center
131                dist = np.linalg.norm(diff)
132
133                if dist > max_displacement:
134                    diff = diff / dist * max_displacement
135                    optimized_path[i] = center + diff
136
137        return optimized_path

139    def get_line_distance_and_angle_diff(self):
140        """
141        Returns:
142        1. line_distance (float): Distance to closest point
143        2. angle_diff (float): Orientation error
144        3. closest_idx (int): The index of the closest point on the track
145        4. is_backwards (bool): True if the car is going backwards along the track
146        """
147        if self.optimal_line is None:
148            return 0.0, 0.0, 0, False
149            
150        car = self.env.unwrapped.car
151        car_pos = np.array(car.hull.position, dtype=np.float32)
152        
153        # 1. Define segments
154        A = self.optimal_line
155        B = np.roll(self.optimal_line, -1, axis=0) 
156        
157        # 2. Vectors
158        AB = B - A              
159        AP = car_pos - A        
160        
161        # 3. Project AP onto AB
162        dot_prod = np.sum(AP * AB, axis=1)
163        ab_len_sq = np.sum(AB**2, axis=1)
164        
165        # Avoid division by zero
166        t = np.divide(dot_prod, ab_len_sq, out=np.zeros_like(dot_prod), where=ab_len_sq!=0)
167        t = np.clip(t, 0.0, 1.0)
168        
169        # 4. Closest point C on the segment
170        C = A + (AB * t[:, np.newaxis])
171        
172        # 5. Distances
173        diff = car_pos - C
174        dists = np.linalg.norm(diff, axis=1)
175        
176        # 6. Find Closest Index
177        closest_idx = np.argmin(dists)
178        line_distance = dists[closest_idx]
179        
180        # 7. Angle Calculation and Direction Check
181        car_vel = car.hull.linearVelocity
182        car_direction = np.array([car_vel.x, car_vel.y], dtype=np.float32)
183        car_speed = np.linalg.norm(car_direction)
184        
185        angle_diff = 0.0
186        is_backwards = False
187        
188        if car_speed > 0.1:
189            car_direction_norm = car_direction / car_speed
190            tangent = AB[closest_idx]
191            tangent_norm = tangent / np.linalg.norm(tangent)
192            
193            # Dot product tells us if we're going forward or backward
194            # Positive = same direction, Negative = opposite direction
195            direction_dot = np.dot(car_direction_norm, tangent_norm)
196            
197            # If dot product is negative, car is going backwards
198            is_backwards = direction_dot < 0
199            
200            # Calculate angle difference (always positive)
201            dot_product = np.clip(direction_dot, -1.0, 1.0)
202            angle_diff = np.arccos(abs(dot_product))
203        
204        return line_distance, angle_diff, closest_idx, is_backwards

206    def get_lateral_velocity(self):
207        """Calculates the lateral (sideways) component of the car's velocity.
208
209        This is used to detect drifting behavior. The method projects the car's velocity
210        vector onto the vector perpendicular to the car's heading direction.
211
212        Returns:
213            np.ndarray: A 2D vector representing the lateral velocity in world coordinates.
214                Returns [0.0, 0.0] if the car's speed is below 0.1 to avoid numerical issues.
215        """
216        car = self.env.unwrapped.car
217        vel = car.hull.linearVelocity
218        speed = np.linalg.norm([vel.x, vel.y])
219        
220        if speed < 0.1:
221            return np.array([0.0, 0.0])
222        
223        vel_norm = np.array([vel.x, vel.y]) / speed
224        
225        # Perpendicular vector to velocity
226        lateral_dir = np.array([-vel_norm[1], vel_norm[0]])
227        
228        lateral_velocity = np.dot(np.array([vel.x, vel.y]), lateral_dir) * lateral_dir
229        return lateral_velocity

249    def process_observation(self, observation):
250        """Full observation processing pipeline: Crop HUD -> Grayscale -> Add Channel Dim.
251
252        Converts the raw RGB observation into the format expected by the CNN policy.
253        The pipeline consists of:
254        1. Cropping the HUD from the bottom of the frame.
255        2. Converting RGB to grayscale to reduce dimensionality.
256        3. Adding a channel dimension for compatibility with Stable-Baselines3.
257
258        Grayscale is used instead of RGB to reduce computational cost while preserving
259        important visual features like road boundaries and curbs.
260
261        Args:
262            observation (np.ndarray): The raw observation from the environment with shape (96, 96, 3).
263
264        Returns:
265            np.ndarray: The processed observation with shape (84, 96, 1).
266        """
267        # 1. Crop HUD
268        obs = self.remove_observation_hud(observation)
269        
270        # 2. Grayscale
271        # IMPORTANT: Do not use binary thresholding. 
272        # We need the gray shades to see the red/white curbs.
273        gray = cv2.cvtColor(obs, cv2.COLOR_RGB2GRAY)
274        
275        # 3. Add Channel Dimension for SB3 (H, W, 1)
276        return np.expand_dims(gray, axis=-1)

278    def step(self, action):
279        """Executes one step in the environment with custom processing.
280
281        This method:
282        1. Executes the action in the wrapped environment.
283        2. Processes the observation (crop HUD, convert to grayscale).
284        3. Updates internal counters (still steps, offroad steps).
285        4. Checks for early stopping conditions (stuck, too much offroad time).
286        5. Applies custom reward shaping if enabled.
287
288        Args:
289            action (np.ndarray): The action to execute, typically a 3-element array
290                [steering, gas, brake] for continuous control.
291
292        Returns:
293            tuple: A tuple containing:
294                - processed_obs (np.ndarray): The processed observation with shape (84, 96, 1).
295                - reward (float): The modified reward after applying custom shaping.
296                - done (bool): Whether the episode has ended (terminal state).
297                - truncated (bool): Whether the episode was truncated (time limit/early stop).
298                - info (dict): Auxiliary diagnostic information from the environment.
299        """
300        observation, reward, done, truncated, info = self.env.step(action)
301        
302        # Process Visuals
303        processed_obs = self.process_observation(observation)
304        
305        # --- CRITICAL FIX: Process Terminal Observation ---
306        # If the episode ends (truncated/done), SB3 looks at 'terminal_observation'.
307        # We must ensure this matches our observation space (84, 96, 1).
308        if "terminal_observation" in info:
309            info["terminal_observation"] = self.process_observation(info["terminal_observation"])
310        
311        # --- SIMPLIFIED REWARD SHAPING ---
312        # We trust the environment's native tile-based reward system.
313        
314        car = self.env.unwrapped.car
315        speed = np.linalg.norm(car.hull.linearVelocity)
316
317        # Track still and offroad steps for early stopping
318        if speed < cfg.STILL_SPEED_THRESHOLD:
319            self.consecutive_still_steps += 1
320        else:
321            self.consecutive_still_steps = 0
322        
323        # Count wheels off road (a wheel is off-road if it has no tiles beneath it)
324        n_offroad_wheels = sum(1 for wheel in car.wheels if len(wheel.tiles) == 0)
325        if n_offroad_wheels >= 4:
326            self.consecutive_offroad_steps += 1
327        else:
328            self.consecutive_offroad_steps = 0
329        
330        # Early stopping if stuck (but no reward penalty when use_additional_rewards=False)
331        if self.consecutive_still_steps > cfg.MAX_STILL_STEPS:
332            truncated = True
333            if self.use_additional_rewards:
334                reward -= cfg.TRUNCATION_PENALTY # Punishment for giving up
335        
336        # Apply reward modifications only if enabled
337        if self.use_additional_rewards:
338            # 1. Wake up call: Penalty for not moving
339            if speed < cfg.STILL_SPEED_THRESHOLD:
340                reward -= cfg.STILL_PENALTY
341                
342            # 2. Completion Bonus
343            if info.get("lap_finished"):
344                reward += cfg.LAP_FINISH_BONUS
345
346        reward = self.apply_additional_rewards(action, reward)
347        
348        self.last_action = action.copy()
349        
350        return processed_obs, reward, done, truncated, info

352    def apply_additional_rewards(self, action, reward):   
353        """Calculates and applies auxiliary rewards and penalties.
354
355        Modifies the base environment reward based on the agent's behavior to encourage
356        optimal racing performance. The modifications include:
357        - Penalties for off-road driving (per wheel).
358        - Penalties for drifting (excessive lateral velocity).
359        - Penalties for wiggling (rapid steering changes).
360        - Penalties for sharp steering at high speeds.
361        - Rewards for staying close to the optimal racing line.
362        - Rewards for maintaining correct orientation relative to the track.
363        - Heavy penalties for driving backwards.
364
365        All reward weights are configured in config.py.
366
367        Args:
368            action (np.ndarray): The action taken in this step.
369            reward (float): The base reward from the environment.
370
371        Returns:
372            float: The modified reward after applying all auxiliary components.
373        """
374        if not self.use_additional_rewards:
375            return reward
376
377        car = self.env.unwrapped.car
378        speed = np.linalg.norm(car.hull.linearVelocity)
379        n_offroad_wheels = sum(1 for wheel in car.wheels if len(wheel.tiles) == 0)
380        
381        if self.offroad_penalty:
382            if n_offroad_wheels > 0:
383                reward -= n_offroad_wheels * cfg.OFFROAD_WHEEL_PENALTY
384        
385        if self.drift_penalty:
386            lateral_velocity = self.get_lateral_velocity()
387            drift_amount = np.linalg.norm(lateral_velocity)
388            if drift_amount > cfg.DRIFT_THRESHOLD:
389                reward -= cfg.DRIFT_PENALTY * (drift_amount / (speed + 1e-5))
390                
391        if self.wiggle_penalty:
392            steering = action[0]
393            wiggle = abs(steering - self.last_action[0]) if self.last_action is not None else 0.0
394            if wiggle > cfg.WIGGLE_THRESHOLD:
395                reward -= cfg.WIGGLE_PENALTY * steering
396        
397        # High-speed steering penalty: penalize sharp steering at high speeds
398        # This teaches the agent that high speed requires gentle steering
399        steering = abs(action[0])
400        if self.fast_steering_penalty and steering > 0.1:  # Only penalize if actually steering
401            speed_factor = min(speed / cfg.TARGET_SPEED, 1.0)
402            high_speed_steering_penalty = cfg.HIGH_SPEED_STEERING_PENALTY * steering * speed_factor
403            reward -= high_speed_steering_penalty
404        
405        if self.line_distance_reward or self.line_angle_reward:
406            line_distance, angle_diff, closest_idx, is_backwards = self.get_line_distance_and_angle_diff()
407            
408            # CRITICAL: Heavy penalty for going backwards
409            # This prevents the agent from exploiting the tile reward by going backwards
410            if is_backwards:
411                # Scale penalty with speed - faster backwards = worse penalty
412                backwards_penalty = (cfg.MAX_LINE_DISTANCE_REWARD + cfg.MAX_LINE_ANGLE_REWARD) * 2.0
413                reward -= backwards_penalty * min(speed / cfg.TARGET_SPEED, 1.5)
414            else:
415                # Only give rewards if going in the correct direction
416                angle_diff_ratio = abs(angle_diff / np.pi)  # Normalize to [0, 1]
417                speed_factor = min(speed / cfg.TARGET_SPEED, 1.0)
418                
419                if self.line_distance_reward:
420                    if line_distance < cfg.LINE_SAFE_DISTANCE:
421                        # Max reward when within safe distance
422                        distance_reward = cfg.MAX_LINE_DISTANCE_REWARD * speed_factor
423                    else:
424                        # Gaussian decay beyond safe distance
425                        distance_reward = np.exp(-(line_distance**2) / (2 * (cfg.LINE_DISTANCE_DECAY**2)))
426                        distance_reward *= cfg.MAX_LINE_DISTANCE_REWARD * speed_factor
427                    reward += distance_reward
428                    
429                if self.line_angle_reward:
430                    if angle_diff < np.radians(cfg.LINE_SAFE_ANGLE):
431                        # Max reward when within safe angle
432                        angle_reward = cfg.MAX_LINE_ANGLE_REWARD * speed_factor
433                    else:
434                        angle_reward = np.exp(-(angle_diff_ratio**2) / (2 * (cfg.LINE_ANGLE_DECAY**2)))
435                        angle_reward *= cfg.MAX_LINE_ANGLE_REWARD * speed_factor
436                    reward += angle_reward
437        
438        return reward

461class CustomRepeatWrapper(gym.Wrapper):
462    """A wrapper that repeats an action for a specified number of frames.
463
464    This is used to reduce the temporal resolution of the environment, allowing
465    the agent to make decisions less frequently while the physics simulation
466    advances multiple steps. This stabilizes training and improves sample efficiency
467    by reducing the effective action space complexity.
468
469    Attributes:
470        repeat (int): The number of times to repeat each action.
471    """
472    def __init__(self, env, repeat=cfg.ACTION_REPEAT):
473        """Initialize the CustomRepeatWrapper.
474
475        Args:
476            env (gym.Env): The environment to wrap.
477            repeat (int, optional): Number of frames to repeat each action.
478                Defaults to cfg.ACTION_REPEAT (typically 2).
479        """
480        super().__init__(env)
481        self.repeat = repeat
482
483    def reset(self, **kwargs):
484        """Resets the environment.
485
486        Args:
487            **kwargs: Additional arguments for the reset method.
488
489        Returns:
490            tuple: (observation, info) from the wrapped environment.
491        """
492        return self.env.reset(**kwargs)
493
494    def step(self, action):
495        """Executes the action repeatedly for `self.repeat` steps.
496
497        Accumulates rewards across the repeated steps. If the episode ends
498        during the repetition (terminated or truncated), the loop breaks early.
499
500        Args:
501            action (np.ndarray): The action to repeat for multiple frames.
502
503        Returns:
504            tuple: A tuple containing:
505                - obs (np.ndarray): The final observation after all repetitions.
506                - total_reward (float): The sum of rewards from all repeated steps.
507                - done (bool): Whether the episode terminated.
508                - truncated (bool): Whether the episode was truncated.
509                - info (dict): Info dictionary from the last step.
510        """
511        total_reward = 0.0
512        done = False
513        truncated = False
514        info = {}
515        
516        for _ in range(self.repeat):
517            obs, rew, terminated, truncated_step, info = self.env.step(action)
518            total_reward += rew
519            if terminated or truncated_step:
520                done = terminated
521                truncated = truncated_step
522                break
523        
524        return obs, total_reward, done, truncated, info

 92    def get_optimal_line(self, iterations=200):
 93        """Calculates an optimal racing line using Laplacian smoothing on track center points.
 94
 95        The algorithm iteratively smooths the track path while constraining points to stay
 96        within the track boundaries. This creates a "racing line" that cuts corners and
 97        optimizes the path through the track.
 98
 99        Args:
100            iterations (int, optional): Number of smoothing iterations. Higher values produce
101                smoother lines but may be computationally expensive. Defaults to 200.
102
103        Returns:
104            np.ndarray: An array of shape (N, 2) containing the (x, y) coordinates of the
105                optimal racing line, where N is the number of track points.
106        """
107        raw_track = self.env.unwrapped.track
108        path = np.array([[p[2], p[3]] for p in raw_track])
109        
110        # Keep the line well within track boundaries
111        SAFE_MARGIN = 1.5  # More conservative margin
112        max_displacement = TRACK_WIDTH - SAFE_MARGIN
113
114        optimized_path = np.copy(path)
115        num_points = len(path)
116
117        for iteration in range(iterations):
118            for i in range(num_points):
119                prev_idx = (i - 1) % num_points
120                next_idx = (i + 1) % num_points
121                
122                # Smoothing: pull towards midpoint of neighbors
123                midpoint = (optimized_path[prev_idx] + optimized_path[next_idx]) / 2
124                # Use more aggressive smoothing in later iterations
125                blend_factor = 0.3 if iteration < iterations // 2 else 0.5
126                optimized_path[i] = optimized_path[i] * (1 - blend_factor) + midpoint * blend_factor
127
128                # Constraint: enforce maximum distance from track center
129                center = path[i]
130                current = optimized_path[i]
131                diff = current - center
132                dist = np.linalg.norm(diff)
133
134                if dist > max_displacement:
135                    diff = diff / dist * max_displacement
136                    optimized_path[i] = center + diff
137
138        return optimized_path

140    def get_line_distance_and_angle_diff(self):
141        """
142        Returns:
143        1. line_distance (float): Distance to closest point
144        2. angle_diff (float): Orientation error
145        3. closest_idx (int): The index of the closest point on the track
146        4. is_backwards (bool): True if the car is going backwards along the track
147        """
148        if self.optimal_line is None:
149            return 0.0, 0.0, 0, False
150            
151        car = self.env.unwrapped.car
152        car_pos = np.array(car.hull.position, dtype=np.float32)
153        
154        # 1. Define segments
155        A = self.optimal_line
156        B = np.roll(self.optimal_line, -1, axis=0) 
157        
158        # 2. Vectors
159        AB = B - A              
160        AP = car_pos - A        
161        
162        # 3. Project AP onto AB
163        dot_prod = np.sum(AP * AB, axis=1)
164        ab_len_sq = np.sum(AB**2, axis=1)
165        
166        # Avoid division by zero
167        t = np.divide(dot_prod, ab_len_sq, out=np.zeros_like(dot_prod), where=ab_len_sq!=0)
168        t = np.clip(t, 0.0, 1.0)
169        
170        # 4. Closest point C on the segment
171        C = A + (AB * t[:, np.newaxis])
172        
173        # 5. Distances
174        diff = car_pos - C
175        dists = np.linalg.norm(diff, axis=1)
176        
177        # 6. Find Closest Index
178        closest_idx = np.argmin(dists)
179        line_distance = dists[closest_idx]
180        
181        # 7. Angle Calculation and Direction Check
182        car_vel = car.hull.linearVelocity
183        car_direction = np.array([car_vel.x, car_vel.y], dtype=np.float32)
184        car_speed = np.linalg.norm(car_direction)
185        
186        angle_diff = 0.0
187        is_backwards = False
188        
189        if car_speed > 0.1:
190            car_direction_norm = car_direction / car_speed
191            tangent = AB[closest_idx]
192            tangent_norm = tangent / np.linalg.norm(tangent)
193            
194            # Dot product tells us if we're going forward or backward
195            # Positive = same direction, Negative = opposite direction
196            direction_dot = np.dot(car_direction_norm, tangent_norm)
197            
198            # If dot product is negative, car is going backwards
199            is_backwards = direction_dot < 0
200            
201            # Calculate angle difference (always positive)
202            dot_product = np.clip(direction_dot, -1.0, 1.0)
203            angle_diff = np.arccos(abs(dot_product))
204        
205        return line_distance, angle_diff, closest_idx, is_backwards

353    def apply_additional_rewards(self, action, reward):   
354        """Calculates and applies auxiliary rewards and penalties.
355
356        Modifies the base environment reward based on the agent's behavior to encourage
357        optimal racing performance. The modifications include:
358        - Penalties for off-road driving (per wheel).
359        - Penalties for drifting (excessive lateral velocity).
360        - Penalties for wiggling (rapid steering changes).
361        - Penalties for sharp steering at high speeds.
362        - Rewards for staying close to the optimal racing line.
363        - Rewards for maintaining correct orientation relative to the track.
364        - Heavy penalties for driving backwards.
365
366        All reward weights are configured in config.py.
367
368        Args:
369            action (np.ndarray): The action taken in this step.
370            reward (float): The base reward from the environment.
371
372        Returns:
373            float: The modified reward after applying all auxiliary components.
374        """
375        if not self.use_additional_rewards:
376            return reward
377
378        car = self.env.unwrapped.car
379        speed = np.linalg.norm(car.hull.linearVelocity)
380        n_offroad_wheels = sum(1 for wheel in car.wheels if len(wheel.tiles) == 0)
381        
382        if self.offroad_penalty:
383            if n_offroad_wheels > 0:
384                reward -= n_offroad_wheels * cfg.OFFROAD_WHEEL_PENALTY
385        
386        if self.drift_penalty:
387            lateral_velocity = self.get_lateral_velocity()
388            drift_amount = np.linalg.norm(lateral_velocity)
389            if drift_amount > cfg.DRIFT_THRESHOLD:
390                reward -= cfg.DRIFT_PENALTY * (drift_amount / (speed + 1e-5))
391                
392        if self.wiggle_penalty:
393            steering = action[0]
394            wiggle = abs(steering - self.last_action[0]) if self.last_action is not None else 0.0
395            if wiggle > cfg.WIGGLE_THRESHOLD:
396                reward -= cfg.WIGGLE_PENALTY * steering
397        
398        # High-speed steering penalty: penalize sharp steering at high speeds
399        # This teaches the agent that high speed requires gentle steering
400        steering = abs(action[0])
401        if self.fast_steering_penalty and steering > 0.1:  # Only penalize if actually steering
402            speed_factor = min(speed / cfg.TARGET_SPEED, 1.0)
403            high_speed_steering_penalty = cfg.HIGH_SPEED_STEERING_PENALTY * steering * speed_factor
404            reward -= high_speed_steering_penalty
405        
406        if self.line_distance_reward or self.line_angle_reward:
407            line_distance, angle_diff, closest_idx, is_backwards = self.get_line_distance_and_angle_diff()
408            
409            # CRITICAL: Heavy penalty for going backwards
410            # This prevents the agent from exploiting the tile reward by going backwards
411            if is_backwards:
412                # Scale penalty with speed - faster backwards = worse penalty
413                backwards_penalty = (cfg.MAX_LINE_DISTANCE_REWARD + cfg.MAX_LINE_ANGLE_REWARD) * 2.0
414                reward -= backwards_penalty * min(speed / cfg.TARGET_SPEED, 1.5)
415            else:
416                # Only give rewards if going in the correct direction
417                angle_diff_ratio = abs(angle_diff / np.pi)  # Normalize to [0, 1]
418                speed_factor = min(speed / cfg.TARGET_SPEED, 1.0)
419                
420                if self.line_distance_reward:
421                    if line_distance < cfg.LINE_SAFE_DISTANCE:
422                        # Max reward when within safe distance
423                        distance_reward = cfg.MAX_LINE_DISTANCE_REWARD * speed_factor

106        """
107        raw_track = self.env.unwrapped.track
108        path = np.array([[p[2], p[3]] for p in raw_track])
109        
110        # Keep the line well within track boundaries
111        SAFE_MARGIN = 1.5  # More conservative margin
112        max_displacement = TRACK_WIDTH - SAFE_MARGIN
113

115        num_points = len(path)
116
117        for iteration in range(iterations):
118            for i in range(num_points):
119                prev_idx = (i - 1) % num_points
120                next_idx = (i + 1) % num_points
121                
122                # Smoothing: pull towards midpoint of neighbors
123                midpoint = (optimized_path[prev_idx] + optimized_path[next_idx]) / 2
124                # Use more aggressive smoothing in later iterations
125                blend_factor = 0.3 if iteration < iterations // 2 else 0.5
126                optimized_path[i] = optimized_path[i] * (1 - blend_factor) + midpoint * blend_factor
127

129                center = path[i]
130                current = optimized_path[i]
131                diff = current - center
132                dist = np.linalg.norm(diff)
133
134                if dist > max_displacement:
135                    diff = diff / dist * max_displacement

152        car_pos = np.array(car.hull.position, dtype=np.float32)
153        
154        # 1. Define segments
155        A = self.optimal_line
156        B = np.roll(self.optimal_line, -1, axis=0) 
157        
158        # 2. Vectors
159        AB = B - A              
160        AP = car_pos - A        
161        
162        # 3. Project AP onto AB
163        dot_prod = np.sum(AP * AB, axis=1)
164        ab_len_sq = np.sum(AB**2, axis=1)
165

169

176        
177        # 6. Find Closest Index
178        closest_idx = np.argmin(dists)
179        line_distance = dists[closest_idx]

355
356        Modifies the base environment reward based on the agent's behavior to encourage
357        optimal racing performance. The modifications include:
358        - Penalties for off-road driving (per wheel).
359        - Penalties for drifting (excessive lateral velocity).
360        - Penalties for wiggling (rapid steering changes).
361        - Penalties for sharp steering at high speeds.
362        - Rewards for staying close to the optimal racing line.
363        - Rewards for maintaining correct orientation relative to the track.
364        - Heavy penalties for driving backwards.
365
366        All reward weights are configured in config.py.
367
368        Args:
369            action (np.ndarray): The action taken in this step.
370            reward (float): The base reward from the environment.
371
372        Returns:
373            float: The modified reward after applying all auxiliary components.
374        """
375        if not self.use_additional_rewards:
376            return reward
377
378        car = self.env.unwrapped.car
379        speed = np.linalg.norm(car.hull.linearVelocity)
380        n_offroad_wheels = sum(1 for wheel in car.wheels if len(wheel.tiles) == 0)
381        
382        if self.offroad_penalty:
383            if n_offroad_wheels > 0:

391

397
