learning

Training infrastructure module for reinforcement learning agent.

This module provides the core training framework for PPO agents in the CarRacing environment. It includes custom callbacks for progress tracking and a Driver class that manages the complete training lifecycle, including model initialization, environment setup, checkpointing, and evaluation.

Key Features: - Automatic frame stacking (4 frames) and image transposition (HWC→CHW) - Custom progress bar with real-time reward tracking - Configurable checkpointing and evaluation callbacks - Support for phased/curriculum training with custom PPO hyperparameters - Proper vectorized environment handling with VecFrameStack and VecTransposeImage

Typical Usage:



from customization import make_vec_envs
from learning import Driver

# Create training and evaluation environments
train_env = make_vec_envs(num_envs=4, use_additional_rewards=True)
eval_env = make_vec_envs(num_envs=1, use_additional_rewards=False)

# Initialize driver and train
driver = Driver(train_env, eval_env=eval_env, training_steps=1_000_000)
driver.train()

 82            total_timesteps (int): Total number of environment steps for the
 83                training run. Used to configure the progress bar range.

 92    def _on_training_start(self):
 93        """Initialize the progress bar when training begins.
 94        
 95        Called automatically by Stable-Baselines3 at the start of training.
 96        Creates the tqdm progress bar with appropriate total timesteps.
 97        """
 98        self.pbar = tqdm(total=self.total_timesteps, desc="Training", unit="step")

100    def _on_step(self):
101        """Update progress bar on each training step.
102        
103        Called after every environment step during training. Updates the progress
104        bar by the number of parallel environments and refreshes the displayed
105        statistics (mean reward and best reward) if new episodes have completed.
106        
107        Returns:
108            bool: Always returns True to continue training.
109        """
110        self.pbar.update(self.training_env.num_envs)
111        
112        if len(self.model.ep_info_buffer) > 0:
113            mean_reward = np.mean([info['r'] for info in self.model.ep_info_buffer])
114            
115            if mean_reward != self.last_mean_reward:
116                self.pbar.set_postfix({
117                    "Mean Reward": f"{mean_reward:.1f}",
118                    "Best": f"{self.last_mean_reward:.1f}"
119                })
120                self.last_mean_reward = mean_reward
121            
122        return True

124    def _on_training_end(self):
125        """Close the progress bar when training completes.
126        
127        Called automatically by Stable-Baselines3 at the end of training.
128        Ensures the progress bar is properly closed and cleaned up.
129        """
130        if self.pbar:
131            self.pbar.close()

212    def __init__(self, vec_env, eval_env=None, algorithm="PPO", training_steps=cfg.DEBUGGING_TIMESTEPS, save_freq=cfg.SAVE_FREQ, 
213                 checkpoint_dir=cfg.CHECKPOINT_DIR, log_dir=cfg.LOG_DIR, 
214                 best_model_dir=cfg.BEST_MODEL_DIR, eval_freq=cfg.EVAL_FREQ, n_eval_episodes=cfg.N_EVAL_EPISODES,
215                 custom_ppo_params=None):
216        """Initialize the Driver with environments and training configuration.
217        
218        Sets up the training infrastructure including environment preprocessing,
219        model initialization, and callback configuration. Automatically applies
220        VecFrameStack and VecTransposeImage to ensure proper observation shapes.
221        
222        Args:
223            vec_env (VecEnv): Vectorized training environment. Will be wrapped with
224                VecFrameStack and VecTransposeImage if not already applied.
225            eval_env (VecEnv, optional): Vectorized evaluation environment. If provided,
226                will also be wrapped with VecFrameStack and VecTransposeImage to match
227                the training environment's preprocessing. Defaults to None.
228            algorithm (str, optional): RL algorithm to use. Currently only "PPO" is
229                supported. Defaults to "PPO".
230            training_steps (int, optional): Total number of physics timesteps for training.
231                Will be divided by ACTION_REPEAT to get agent steps. Defaults to
232                cfg.DEBUGGING_TIMESTEPS.
233            save_freq (int, optional): Frequency (in agent steps) for saving model
234                checkpoints. If None, automatically set to min(training_steps // 10, 50000).
235                Defaults to cfg.SAVE_FREQ.
236            checkpoint_dir (str, optional): Directory path for saving model checkpoints.
237                Defaults to cfg.CHECKPOINT_DIR.
238            log_dir (str, optional): Directory path for TensorBoard logs.
239                Defaults to cfg.LOG_DIR.
240            best_model_dir (str, optional): Directory path for saving the best model
241                (based on evaluation performance). Defaults to cfg.BEST_MODEL_DIR.
242            eval_freq (int, optional): Frequency (in agent steps) for running evaluation.
243                If None, set equal to save_freq. Defaults to cfg.EVAL_FREQ.
244            n_eval_episodes (int, optional): Number of episodes to run during each
245                evaluation. Defaults to cfg.N_EVAL_EPISODES.
246            custom_ppo_params (dict, optional): Dictionary of PPO hyperparameters to
247                override defaults from cfg.PPO_PARAMS. Useful for phase-specific
248                configurations in curriculum learning. Example: {"learning_rate": 0.0001,
249                "ent_coef": 0.01}. Defaults to None.
250        
251        Raises:
252            ValueError: If algorithm is not "PPO".
253        
254        Note:
255            - Training steps are converted from physics steps to agent steps by
256              dividing by cfg.ACTION_REPEAT (typically 2).
257            - Both save_freq and eval_freq are adjusted by the number of parallel
258              environments when setting up callbacks.
259            - All output directories are created automatically if they don't exist.
260        """
261        
262        self.vec_env = vec_env
263        self.eval_env = eval_env
264        
265        # Calculate Agent Steps from Physics Steps
266        self.training_steps = training_steps // cfg.ACTION_REPEAT
267        
268        self.save_freq = save_freq
269        self.eval_freq = eval_freq
270        self.n_eval_episodes = n_eval_episodes
271        self.checkpoint_dir = checkpoint_dir
272        self.log_dir = log_dir
273        self.best_model_dir = best_model_dir
274        
275        MAXIMUM_SAVE_FREQ = 50000
276        
277        if self.save_freq is None:
278            self.save_freq = min(training_steps // 10, MAXIMUM_SAVE_FREQ)
279        
280        if self.eval_freq is None:
281            self.eval_freq = self.save_freq
282
283        # Stack 4 frames for Training Env
284        if not isinstance(vec_env, VecFrameStack):
285            self.vec_env = VecFrameStack(vec_env, n_stack=4)
286        
287        # FIX: Explicitly Transpose Training Env to ensure consistent shape (C, H, W)
288        # This prevents PPO from doing it implicitly and prevents the (6, 84, 96) shape error
289        if not isinstance(self.vec_env, VecTransposeImage):
290            self.vec_env = VecTransposeImage(self.vec_env)
291            
292        # FIX: Ensure Eval Env is stacked AND transposed to match Training Env
293        if self.eval_env is not None:
294            if not isinstance(self.eval_env, VecFrameStack):
295                self.eval_env = VecFrameStack(self.eval_env, n_stack=4)
296            
297            # CRITICAL: PPO auto-wraps training env in VecTransposeImage (HWC->CHW).
298            # We must manually wrap eval_env to match, otherwise we get a type warning
299            # and the model receives scrambled shapes during eval.
300            if not isinstance(self.eval_env, VecTransposeImage):
301                self.eval_env = VecTransposeImage(self.eval_env)
302            
303        self.algorithm = algorithm.lower()
304        
305        # Create directories
306        os.makedirs(checkpoint_dir, exist_ok=True)
307        os.makedirs(log_dir, exist_ok=True)
308        os.makedirs(best_model_dir, exist_ok=True)
309        
310        # --- MODEL SETUP ---
311        if self.algorithm == "ppo":
312            # Start with default params, then override with custom ones if provided
313            self.model_params = cfg.PPO_PARAMS.copy()
314            if custom_ppo_params:
315                # Merge custom params (phase-specific) with defaults
316                self.model_params.update(custom_ppo_params)
317            
318            self.model_params["env"] = self.vec_env
319            self.model_params["tensorboard_log"] = self.log_dir
320            self.model = PPO(**self.model_params)
321        
322        # Setup callbacks
323        self.checkpoint_callback = None
324        self.eval_callback = None
325        self._setup_callbacks()

327    def _setup_callbacks(self):
328        """Configure checkpoint and evaluation callbacks for training.
329        
330        Creates CheckpointCallback for periodic model saving and EvalCallback
331        for evaluation (if eval_env is provided). Frequencies are adjusted by
332        the number of parallel environments to ensure correct timestep intervals.
333        
334        The evaluation callback saves the best-performing model based on mean
335        reward across evaluation episodes.
336        """
337        num_envs = self.vec_env.num_envs
338        self.checkpoint_callback = CheckpointCallback(
339            save_freq=self.save_freq // num_envs,
340            save_path=self.checkpoint_dir,
341            name_prefix="ppo_model"
342        )
343        if self.eval_env is not None:
344            self.eval_callback = EvalCallback(
345                self.eval_env,
346                best_model_save_path=self.best_model_dir,
347                log_path=os.path.join(self.log_dir, "eval"),
348                eval_freq=max(self.eval_freq // num_envs, 1),
349                n_eval_episodes=self.n_eval_episodes,
350                deterministic=True,
351                render=False
352            )

354    def train(self, steps=None, reset_num_timesteps=True):
355        """Execute the training loop for the specified number of steps.
356        
357        Trains the PPO model using the configured environment and callbacks.
358        Displays an enhanced progress bar with real-time reward tracking during
359        training. Automatically saves checkpoints and runs evaluations based on
360        the configured frequencies.
361        
362        Args:
363            steps (int, optional): Number of agent steps to train for. If None,
364                uses self.training_steps (set during initialization). Defaults to None.
365            reset_num_timesteps (bool, optional): Whether to reset the timestep
366                counter to zero. Set to False when continuing training from a loaded
367                checkpoint. Defaults to True.
368        
369        Example:
370            ```python
371            # Initial training
372            driver.train(steps=1_000_000)
373            
374            # Continue training from checkpoint
375            driver.load_model("path/to/checkpoint.zip")
376            driver.train(steps=500_000, reset_num_timesteps=False)
377            ```
378        
379        Note:
380            The steps parameter refers to agent steps, not physics steps.
381            Physics steps = agent steps × ACTION_REPEAT.
382        """
383        """Train the model for specified steps"""
384        if steps is None:
385            steps = self.training_steps
386            
387        callbacks = [self.checkpoint_callback]
388        if self.eval_callback is not None:
389            callbacks.append(self.eval_callback)
390        
391        # Add Progress Bar
392        progress_callback = EnhancedProgressBarCallback(total_timesteps=steps)
393        callbacks.append(progress_callback)
394            
395        self.model.learn(
396            total_timesteps=steps,
397            callback=callbacks,
398            progress_bar=False, 
399            reset_num_timesteps=reset_num_timesteps
400        )
